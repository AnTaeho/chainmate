## 셈 스크립트(dstats.mjs)
```js
// D 여세 근거: 판 하네스 dump에서 판 안 대국 점수/목표(넘침 비율) 분포와 판 사이 · 판 안 들쭉날쭉
import { readFileSync } from 'node:fs';
const files = process.argv.slice(2);
const runs = files.flatMap((f) => JSON.parse(readFileSync(f, 'utf8')).runs);
const q = (a, p) => { const s = [...a].sort((x, y) => x - y); return s[Math.min(s.length - 1, Math.round(p * (s.length - 1)))]; };
const mean = (a) => a.reduce((s, x) => s + x, 0) / a.length;
const vr = (a) => { const m = mean(a); return a.reduce((s, x) => s + (x - m) ** 2, 0) / (a.length - 1); };
const pc = (x) => (100 * x).toFixed(1) + '%';
const played = runs.map((r) => r.log.filter((x) => !x.skipped && x.target > 0));
const all = played.flat();
const won = all.filter((x) => x.won), lost = all.filter((x) => !x.won);
const ratio = (x) => x.score / x.target;
console.log(`판 ${runs.length}개 · 둔 대국 ${all.length}개(이김 ${won.length} · 짐 ${lost.length})`);
const rw = won.map(ratio);
console.log(`이긴 대국 점수/목표: p10 ${q(rw, .1).toFixed(2)} · p25 ${q(rw, .25).toFixed(2)} · p50 ${q(rw, .5).toFixed(2)} · p75 ${q(rw, .75).toFixed(2)} · p90 ${q(rw, .9).toFixed(2)}`);
console.log(`  넘침 층(이긴 대국 중): <1.25 ${pc(rw.filter((r) => r < 1.25).length / rw.length)} · 1.25~2 ${pc(rw.filter((r) => r >= 1.25 && r < 2).length / rw.length)} · 2~5 ${pc(rw.filter((r) => r >= 2 && r < 5).length / rw.length)} · 5~10 ${pc(rw.filter((r) => r >= 5 && r < 10).length / rw.length)} · ≥10 ${pc(rw.filter((r) => r >= 10).length / rw.length)}`);
const rl = lost.map(ratio);
console.log(`진 대국 점수/목표: p10 ${q(rl, .1).toFixed(2)} · p25 ${q(rl, .25).toFixed(2)} · p50 ${q(rl, .5).toFixed(2)} · p75 ${q(rl, .75).toFixed(2)} · p90 ${q(rl, .9).toFixed(2)}`);
// 판 안 · 판 사이: log(점수/목표)를 관 · 종류 평균을 뺀 뒤, 판 평균(사이)과 판 안 흩어짐(안)으로 나눔
const key = (x) => `${x.ante}:${x.kind}`;
const byKey = {};
for (const x of all) (byKey[key(x)] ||= []).push(Math.log(Math.max(1e-3, ratio(x))));
const km = Object.fromEntries(Object.entries(byKey).map(([k, v]) => [k, mean(v)]));
const resid = played.map((rs) => rs.map((x) => Math.log(Math.max(1e-3, ratio(x))) - km[key(x)])).filter((a) => a.length >= 3);
const flat = resid.flat(), tot = vr(flat);
const within = mean(resid.map(vr));
console.log(`log(점수/목표) 관 · 종류 평균을 뺀 분산 ${tot.toFixed(3)}: 판 안 ${within.toFixed(3)}(${pc(within / tot)}) · 판 사이 ${pc(1 - within / tot)}`);
const sds = resid.map((a) => Math.sqrt(vr(a)));
console.log(`판 안 sd(log): p25 ${q(sds, .25).toFixed(2)} · p50 ${q(sds, .5).toFixed(2)} · p75 ${q(sds, .75).toFixed(2)} → 같은 판 안 대국 사이 점수/목표의 전형 폭 ×${Math.exp(q(sds, .5)).toFixed(2)}`);
// 판 안에서 크게 넘친 대국(≥5)과 겨우 넘거나 진 대국(<1.25)이 함께 있는 판
const mixed = played.filter((rs) => rs.some((x) => x.won && ratio(x) >= 5) && rs.some((x) => ratio(x) < 1.25)).length;
console.log(`한 판 안에 ≥×5 이긴 대국과 <×1.25 대국(짐 포함)이 함께 있는 판 ${mixed} / ${runs.length}(${pc(mixed / runs.length)})`);
// 진 대국 바로 앞 대국(같은 판)
const prevs = [], carry = [];
for (const rs of played) rs.forEach((x, i) => {
  if (x.won || i === 0) return;
  const p = rs[i - 1];
  prevs.push(p.won ? ratio(p) : 0);
  // 설계 D(25%, 다음 목표 절반 상한, 지면 사라짐)를 그대로 얹으면 넘었을까 — 다른 것은 그대로 둔 셈
  const c = p.won ? Math.min(0.25 * (p.score - p.target), 0.5 * x.target) : 0;
  carry.push({ c, reach: x.score + c >= x.target, ended: rs === rs && i === rs.length - 1 });
});
console.log(`진 대국 ${prevs.length}개(판 첫 대국 빼고)의 바로 앞 대국 점수/목표(앞도 졌으면 0): p25 ${q(prevs, .25).toFixed(2)} · p50 ${q(prevs, .5).toFixed(2)} · p75 ${q(prevs, .75).toFixed(2)}`);
const r1 = carry.filter((x) => x.reach).length;
console.log(`설계 D를 점수에만 얹었을 때 넘었을 진 대국 ${r1} / ${carry.length}(${pc(r1 / carry.length)}) · 여세 0(앞이 짐) ${carry.filter((x) => x.c === 0).length}개`);
const ends = []; for (const rs of played) { const i = rs.length - 1, x = rs[i]; if (!x.won && i > 0) { const p = rs[i - 1]; const c = p.won ? Math.min(0.25 * (p.score - p.target), 0.5 * x.target) : 0; ends.push(x.score + c >= x.target); } }
console.log(`  그중 판을 끝낸 패배(마지막 대국): ${ends.filter(Boolean).length} / ${ends.length}(${pc(ends.filter(Boolean).length / Math.max(1, ends.length))})`);
const capHit = []; for (const rs of played) rs.forEach((x, i) => { if (i && rs[i - 1].won) capHit.push(0.25 * (rs[i - 1].score - rs[i - 1].target) >= 0.5 * x.target); });
console.log(`앞 대국을 이긴 대국 ${capHit.length}개 중 여세가 상한(다음 목표 절반)에 닿는 몫 ${pc(capHit.filter(Boolean).length / capHit.length)}`);
```
